document.fonts.ready.then(async()=>{
await Promise.all([400,500,700,750].map(weight=>document.fonts.load(weight+' 100px Inter')));
const canvas=document.createElement('canvas'),ctx=canvas.getContext('2d');
for(const svg of document.querySelectorAll('svg.text')){
const text=svg.querySelector('text'),weight=svg.dataset.weight;
ctx.font=weight+' 100px Inter';
const m=ctx.measureText(text.textContent),left=m.actualBoundingBoxLeft,right=m.actualBoundingBoxRight;
svg.setAttribute('viewBox',[-left,-m.actualBoundingBoxAscent,left+right,m.actualBoundingBoxAscent+m.actualBoundingBoxDescent].join(' '));
text.setAttribute('font-size','100');text.setAttribute('font-weight',weight);
}
